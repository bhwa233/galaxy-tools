cmake_minimum_required(VERSION 3.24)
project(spark233_capture LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "The selected ScreenCapture/ImageReader backend requires Windows MSVC.")
endif()
if(NOT DEFINED SPARK233_NATIVE_SOURCE_ROOT)
  message(FATAL_ERROR "Run scripts/native-capture/prepare.ps1 and adapt.ps1; pass their adapted directory.")
endif()
file(TO_CMAKE_PATH "${SPARK233_NATIVE_SOURCE_ROOT}" SPARK233_NATIVE_SOURCE_ROOT)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY MultiThreaded)
set(CMAKE_OBJECT_PATH_MAX 180)
# Decode upstream UTF-8 resource text; adapt.ps1 gives resource files ASCII names.
set(CMAKE_RC_FLAGS "${CMAKE_RC_FLAGS} /c65001")
add_compile_options(/utf-8 /permissive- /W3)
add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE _CRT_SECURE_NO_WARNINGS)

# Preserve the upstream source trees and their licences; CMake replaces only the
# author's absolute Visual Studio include/library paths, not the selected backend.
set(LING "${SPARK233_NATIVE_SOURCE_ROOT}/Ling")
set(CAPTURE "${SPARK233_NATIVE_SOURCE_ROOT}/ScreenCapture/Src")
file(GLOB LING_SOURCES CONFIGURE_DEPENDS "${LING}/src/*.cpp")
file(GLOB_RECURSE YOGA_SOURCES CONFIGURE_DEPENDS "${LING}/yoga/*.cpp")
add_library(spark233_yoga STATIC ${YOGA_SOURCES})
target_include_directories(spark233_yoga PUBLIC "${LING}")
add_library(spark233_ling STATIC ${LING_SOURCES})
target_include_directories(spark233_ling PUBLIC "${LING}" PRIVATE "${LING}/src")
target_precompile_headers(spark233_ling PRIVATE "${LING}/src/pch.h")
target_link_libraries(spark233_ling PUBLIC spark233_yoga windowsapp runtimeobject d3d11 d2d1 dwrite
  user32 shcore dwmapi shell32 shlwapi windowscodecs ole32 oleaut32 gdi32 comdlg32 advapi32 version)

file(GLOB_RECURSE CAPTURE_SOURCES CONFIGURE_DEPENDS "${CAPTURE}/*.cpp" "${CAPTURE}/*.c")
add_executable(ScreenCapture WIN32 ${CAPTURE_SOURCES} "${CAPTURE}/Res/Resource.rc")
target_include_directories(ScreenCapture PRIVATE "${CAPTURE}")
target_include_directories(ScreenCapture PRIVATE "$ENV{VCToolsInstallDir}/atlmfc/include")
target_link_directories(ScreenCapture PRIVATE "$ENV{VCToolsInstallDir}/atlmfc/lib/x64")
target_compile_definitions(ScreenCapture PRIVATE QUIRC_MAX_REGIONS=65534)
target_link_libraries(ScreenCapture PRIVATE spark233_ling comctl32 imm32 version mf mfreadwrite mfplat mfuuid ntdll userenv)
target_link_options(ScreenCapture PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_LIST_DIR}/per-monitor.manifest)

option(SPARK233_BUILD_IMAGE_READER "Build the selected OCR backend when its native dependencies are supplied" OFF)
if(SPARK233_BUILD_IMAGE_READER)
  find_package(OpenCV REQUIRED COMPONENTS core imgproc)
  find_path(ORT_INCLUDE onnxruntime_cxx_api.h REQUIRED HINTS "${ONNXRUNTIME_ROOT}/include")
  find_library(ORT_LIBRARY onnxruntime REQUIRED HINTS "${ONNXRUNTIME_ROOT}/lib")
  set(TINY "${SPARK233_NATIVE_SOURCE_ROOT}/TinyOCR")
  file(GLOB TINY_SOURCES CONFIGURE_DEPENDS "${TINY}/TinyOCR/src/*.cpp" "${TINY}/Clipper2/src/*.cpp")
  add_library(spark233_tinyocr STATIC ${TINY_SOURCES})
  target_include_directories(spark233_tinyocr PUBLIC "${TINY}/TinyOCR/include" "${TINY}/Clipper2/include" "${ORT_INCLUDE}" ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(spark233_tinyocr PUBLIC "${ORT_LIBRARY}" ${OpenCV_LIBS})
  set(READER "${SPARK233_NATIVE_SOURCE_ROOT}/ImageReader/ImageReader")
  file(GLOB READER_SOURCES CONFIGURE_DEPENDS "${READER}/*.cpp")
  add_executable(ImageReader WIN32 ${READER_SOURCES} "${READER}/Res/Resource.rc")
  target_compile_definitions(ImageReader PRIVATE SPARK233_OCR_WORKER)
  target_include_directories(ImageReader PRIVATE "${READER}" "${TINY}")
  target_link_libraries(ImageReader PRIVATE spark233_ling spark233_tinyocr)
  target_link_options(ImageReader PRIVATE /MANIFEST:EMBED /MANIFESTINPUT:${CMAKE_CURRENT_LIST_DIR}/per-monitor.manifest)
  foreach(ORT_DLL onnxruntime.dll onnxruntime_providers_shared.dll)
    if(EXISTS "${ONNXRUNTIME_ROOT}/lib/${ORT_DLL}")
      add_custom_command(TARGET ImageReader POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_if_different
        "${ONNXRUNTIME_ROOT}/lib/${ORT_DLL}" "$<TARGET_FILE_DIR:ImageReader>")
    endif()
  endforeach()
endif()
