cmake_minimum_required(VERSION 3.25)
project(spark233_clipboard_probe LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "The clipboard boundary requires Windows and MSVC")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(spark233_clipboard_boundary boundary.cpp)
target_compile_options(spark233_clipboard_boundary PRIVATE /W4 /utf-8 /EHsc /permissive-)
target_compile_definitions(spark233_clipboard_boundary PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(spark233_clipboard_boundary PRIVATE user32 advapi32 shell32)
add_executable(spark233_clipboard_service service.cpp)
target_compile_options(spark233_clipboard_service PRIVATE /W4 /utf-8 /EHsc /permissive-)
target_compile_definitions(spark233_clipboard_service PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(spark233_clipboard_service PRIVATE user32 advapi32 shell32 windowsapp runtimeobject windowscodecs ole32)
